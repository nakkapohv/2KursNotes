
Операторы:

- FROM - в нём указывается какую сущность выводим (он обязателен, без него запрос на вывод сущности не сработает)
- SELECT - Выводит атрибуты и сущности (обязателен) 
  SELECT * - выводит все атрибуты из сущности
- INSERT INTO - команда для заполнения сущности данными 
- DEFAULT - выводит значение по умолчанию 
- WHERE - когда, необязательный, фильтрует строки по условию

DROP TABLE IF EXISTS Попытки;
DROP TABLE IF EXISTS Нагрузка;
DROP TABLE IF EXISTS Дисциплины_Специальность;
DROP TABLE IF EXISTS Вид_Контроля;
DROP TABLE IF EXISTS Оценивание;
DROP TABLE IF EXISTS Статус;
DROP TABLE IF EXISTS Группа;
DROP TABLE IF EXISTS Студент;
DROP TABLE IF EXISTS Дисциплина;
DROP TABLE IF EXISTS Специальность;
DROP TABLE IF EXISTS Преподаватель;
GO

CREATE TABLE Преподаватель (
    ID SMALLINT IDENTITY(1, 1) PRIMARY KEY NOT NULL,
    ИМЯ NVARCHAR(50) NOT NULL DEFAULT 'PC',
    ФАМИЛИЯ NVARCHAR(50) NOT NULL,
    ОТЧЕСТВО NVARCHAR(50) NULL
);

CREATE TABLE Студент (
    ID SMALLINT IDENTITY(1, 1),
    ИМЯ NVARCHAR(50) NOT NULL,
    ФАМИЛИЯ NVARCHAR(50) NOT NULL,
    ОТЧЕСТВО NVARCHAR(50) NULL,
    НОМЕР_ЗАЧЕТКИ NVARCHAR(15) NOT NULL,
    ДАТА_РОЖДЕНИЯ DATE NOT NULL, 
    CONSTRAINT PK_Студент PRIMARY KEY (ID),
    CONSTRAINT UQ_Студент_Номер_зачетки UNIQUE (НОМЕР_ЗАЧЕТКИ) 
);

CREATE TABLE Специальность (
    ID SMALLINT IDENTITY(1, 1) PRIMARY KEY NOT NULL,
    Название NVARCHAR(150) NOT NULL,
    Квалификация NCHAR(11) NOT NULL
);

CREATE TABLE Дисциплина (
    ID SMALLINT IDENTITY(1, 1) PRIMARY KEY NOT NULL,
    Название NVARCHAR(100) NOT NULL,
    Шифр NVARCHAR(10) NOT NULL
);

CREATE TABLE Оценивание (
    ID TINYINT IDENTITY(1, 1) PRIMARY KEY NOT NULL,
    Результат NVARCHAR(20) NOT NULL
);

CREATE TABLE Статус (
    ID TINYINT IDENTITY(1, 1) PRIMARY KEY NOT NULL,
    Вид NVARCHAR(15) NOT NULL
);

CREATE TABLE Вид_Контроля (
    ID TINYINT IDENTITY(1, 1) PRIMARY KEY NOT NULL,
    Вид NVARCHAR(20) NOT NULL,
    ID_Оценивание TINYINT NULL,
    FOREIGN KEY (ID_Оценивание) REFERENCES Оценивание(ID)
);

CREATE TABLE Дисциплины_Специальность (
    ID INT IDENTITY(1,1) PRIMARY KEY NOT NULL,
    ID_Дисциплина SMALLINT NULL,
    ID_Специальность SMALLINT NULL,
    FOREIGN KEY (ID_Дисциплина) REFERENCES Дисциплина(ID),
    FOREIGN KEY (ID_Специальность) REFERENCES Специальность(ID), 
    CONSTRAINT UQ_Дисциплина_Специальность UNIQUE (ID_Дисциплина, ID_Специальность)
);

CREATE TABLE Группа (
    ID SMALLINT IDENTITY(1, 1) PRIMARY KEY NOT NULL,
    Название NVARCHAR(50) NOT NULL,
    ID_Специальность SMALLINT NULL,
    FOREIGN KEY (ID_Специальность) REFERENCES Специальность(ID)
);

CREATE TABLE Нагрузка (
    ID INT IDENTITY(1, 1) PRIMARY KEY NOT NULL,
    ID_Группа SMALLINT NULL,
    ID_Дисциплина SMALLINT NULL,
    Учебный_год CHAR(9) NOT NULL,
    Семестр CHAR(8) NOT NULL,
    Объем_часов SMALLINT NOT NULL,
    FOREIGN KEY (ID_Группа) REFERENCES Группа(ID),
    FOREIGN KEY (ID_Дисциплина) REFERENCES Дисциплина(ID),
    CONSTRAINT UQ_Нагрузка UNIQUE (ID_Группа, ID_Дисциплина, Семестр),
    CHECK (Семестр >= 1 AND Семестр <= 8),
    CHECK (Учебный_год LIKE '2[0-9][0-9][0-9]/2[0-9][0-9][0-9]') 
);

CREATE TABLE Попытки (
    ID INT IDENTITY(1, 1) PRIMARY KEY NOT NULL,
    ID_Студент SMALLINT NULL,
    ID_Преподаватель SMALLINT NULL,
    ID_Вид_Контроля TINYINT NULL,
    ID_Нагрузка INT NULL,
    Номер_попытки SMALLINT NOT NULL,
    Тип_пересдачи NVARCHAR(50) NOT NULL,
    Дата_сдачи DATE NOT NULL,
    FOREIGN KEY (ID_Студент) REFERENCES Студент(ID),
    FOREIGN KEY (ID_Преподаватель) REFERENCES Преподаватель(ID),
    FOREIGN KEY (ID_Вид_Контроля) REFERENCES Вид_Контроля(ID),
    FOREIGN KEY (ID_Нагрузка) REFERENCES Нагрузка(ID),
    CONSTRAINT UQ_Попытки UNIQUE (ID_Студент, ID_Вид_Контроля, ID_Нагрузка, Номер_попытки),
    CHECK (Номер_попытки > 0 AND Номер_попытки <= 3),
    );
