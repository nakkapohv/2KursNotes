
ALTER TABLE Название_сущности ADD Название_атрибута  - добавить атрибут

Пример 1: Добавим в сущность Блюдо атрибут "Вес порции (г)"
ALTER TABLE Блюдо
ADD Вес_порции INT NULL;

SELECT * FROM Блюдо

Пример 2: Добавим в Ингредиент атрибут "Калорийность" с ограничением CHECK:

ALTER TABLE Ингредиент
ADD Калорийность SMALLINT NULL
CHECK Калорийность >= 0;
SELECT * FROM Ингредиент

UPDATE Название_Сущности SET Атрибут - изменение атрибута

Пример:

UPDATE Блюдо 
SET Всего_ингредиентов = 1;

UPDATE Блюдо
SET Всего_ингредиентов = (
	Select COUNT (*)
	FROM Состав
	WHERE ID_Блюда = Блюдо.ID
)

При изменении и удалении атрибутов не должно быть противоречия (Если есть противоречие сначала избавляемся от ограничения, потом меняем атрибут и если нужно возвращаем ограничение)
